تمرین 1 — باینری‌کردن تصویر تا فقط سه رقم بماند

این تصویر را باینری کنید طوری که فقط سه عددِ نوشته‌شده باقی بماند. (راهنمایی: استفاده از GaussianBlur با فیلترسایز بزرگ برای حذف نویز ضروری است.)

q8.ipynbمقایسه‌ی روش‌های باینری‌سازیPY
import cv2 import matplotlib.pyplot as plt image = cv2.imread("adad.jpg", 0) median = cv2.medianBlur(image, 5) gaussian = cv2.GaussianBlur(image, (5, 5), 0) # مقایسه‌ی نه حالت: (بدون بلور/مدین/گاوسی) × (آستانه‌ی تطبیقی Mean/Gaussian) images = [ image, median, gaussian, cv2.adaptiveThreshold(image, 255, cv2.ADAPTIVE_THRESH_MEAN_C, cv2.THRESH_BINARY, 11, 2), cv2.adaptiveThreshold(median, 255, cv2.ADAPTIVE_THRESH_MEAN_C, cv2.THRESH_BINARY, 11, 2), cv2.adaptiveThreshold(gaussian, 255, cv2.ADAPTIVE_THRESH_MEAN_C, cv2.THRESH_BINARY, 11, 2), cv2.adaptiveThreshold(image, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2), cv2.adaptiveThreshold(median, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2), cv2.adaptiveThreshold(gaussian, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2) ] titles = [ "Original", "Median Blur", "Gaussian Blur", "Original + Adaptive Mean", "Median + Adaptive Mean", "Gaussian + Adaptive Mean", "Original + Adaptive Gaussian", "Median + Adaptive Gaussian", "Gaussian + Adaptive Gaussian" ] plt.figure(figsize=(15, 10)) for i in range(9): plt.subplot(3, 3, i + 1) plt.imshow(images[i], cmap="gray") plt.title(titles[i]) plt.axis("off") plt.tight_layout() plt.show() # مقایسه با Otsu روی همون سه ورودی (خام / مدین / گاوسی) otsu_original = cv2.threshold(image, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)[1] otsu_median = cv2.threshold(median, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)[1] otsu_gaussian = cv2.threshold(gaussian, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)[1] otsu_images = [otsu_original, otsu_median, otsu_gaussian] otsu_titles = ["Original + Otsu", "Median + Otsu", "Gaussian + Otsu"] plt.figure(figsize=(15, 5)) for i in range(3): plt.subplot(1, 3, i + 1) plt.imshow(otsu_images[i], cmap="gray") plt.title(otsu_titles[i]) plt.axis("off") plt.tight_layout() plt.show()

نتیجه

بین روش‌های بررسی‌شده، ترکیب Median Blur + Adaptive Gaussian Thresholding بهترین نتیجه را داد — اعداد را واضح‌تر نگه داشت و نویز پس‌زمینه‌ی کمتری تولید کرد. علتش این است که Adaptive Thresholding برای نواحی مختلف تصویر آستانه‌های متفاوتی در نظر می‌گیرد، پس در قسمت‌هایی که روشنایی به‌خاطر سایه تغییر کرده هم خوب عمل می‌کند — برخلاف Otsu که یک آستانه‌ی سراسری برای کل تصویر پیدا می‌کند و روی این عکس (به‌خاطر سایه و نور نامتوازن) خوب کار نمی‌کند، چون هیستوگرام تصویر دیگر ساده و دوقله‌ای نیست.

تمرین 2 — شمردن ارقام با Connected Components

در ادامه‌ی تمرین قبل، با شمردن تعداد مؤلفه‌های متصل، تعداد ارقام روی کاغذ را بشمارید و چاپ کنید. (راهنمایی: بدیهی است که باید این ۳ عدد سفید و پس‌زمینه سیاه باشد تا بتوانید مؤلفه‌های متصل را حساب کنید.)

q9.ipynbشمارش ارقام با Connected ComponentsPY
import cv2 import matplotlib.pyplot as plt image = cv2.imread("adad.jpg", 0) median = cv2.medianBlur(image, 5) binary = cv2.adaptiveThreshold( median, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2 ) # باید اعداد سفید و پس‌زمینه سیاه باشد تا connectedComponents درست کار کند binary = cv2.bitwise_not(binary) num_labels, labels, stats, centroids = cv2.connectedComponentsWithStats(binary) result = cv2.cvtColor(binary, cv2.COLOR_GRAY2BGR) count = 0 for i in range(1, num_labels): # از 1 شروع می‌شود چون label صفر پس‌زمینه است area = stats[i, cv2.CC_STAT_AREA] x = stats[i, cv2.CC_STAT_LEFT] y = stats[i, cv2.CC_STAT_TOP] width = stats[i, cv2.CC_STAT_WIDTH] height = stats[i, cv2.CC_STAT_HEIGHT] if area > 100 and height > 20: # فیلتر نویزهای کوچک count += 1 cv2.rectangle(result, (x, y), (x + width, y + height), (0,0,255), 2) print("Number of digits:", count) plt.figure(figsize=(15, 5)) plt.subplot(131); plt.imshow(image, cmap="gray"); plt.title("Original"); plt.axis("off") plt.subplot(132); plt.imshow(binary, cmap="gray"); plt.title("Clean"); plt.axis("off") plt.subplot(133); plt.imshow(cv2.cvtColor(result, cv2.COLOR_BGR2RGB)); plt.title("Detected Digits"); plt.axis("off") plt.tight_layout() plt.show()
i
چرا connectivity اینجا مهمه

connectedComponentsWithStats علاوه بر labels، آرایه‌ی stats را هم می‌دهد که مساحت و bounding box هر جزء را از قبل حساب کرده — دیگر لازم نیست جداگانه با Contour این‌ها را دوباره محاسبه کنی. فیلتر area > 100 and height > 20 برای حذف نویزهای ریز باقی‌مانده از لبه‌ی دندانه‌دار آستانه‌گذاری لازم است، وگرنه چند «رقم» اضافه‌ی جعلی شمرده می‌شود.

تمرین 3 — مرکز هر رقم و مستطیل دور رقم بالا-راست

۱. روی عکسی که در دو تمرین قبل استفاده کردیم، مرکز هر عدد را به‌دست آورید و با یک نقطه مشخص کنید.
۲. با عملیات مرتبط با کانتور و همچنین محاسبه‌ی مرکز کانتور، دور سمت‌راست‌ترین و بالاترین عدد (عدد سمت راستِ بالا) مستطیل دربرگیرنده بکشید.

i
نکته‌ی مهم درباره‌ی مختصات تصویر در OpenCV

مبدأ مختصات (0,0) در گوشه‌ی بالا-چپ تصویر است. پس هرچه به سمت بالا برویم، cy کمتر می‌شود؛ هرچه به سمت راست برویم، cx بیشتر می‌شود. برای پیداکردن «بالاترین» رقم باید دنبال کمترین cy باشیم، نه بیشترین.

q10.ipynbمرکز ارقام و رقم بالا-راستPY
import cv2 import matplotlib.pyplot as plt image = cv2.imread("adad.jpg") gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) median = cv2.medianBlur(gray, 5) binary_image = cv2.adaptiveThreshold( median, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2 ) binary_image = cv2.bitwise_not(binary_image) contours, hierarchy = cv2.findContours( binary_image, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE ) contours = [c for c in contours if cv2.contourArea(c) > 100] # حذف نویزهای کوچک def x_cord_contour(contour): M = cv2.moments(contour) return int(M['m10'] / M['m00']) def y_cord_contour(contour): M = cv2.moments(contour) return int(M['m01'] / M['m00']) # رسم مرکز هر رقم for c in contours: M = cv2.moments(c) cx = int(M['m10'] / M['m00']) cy = int(M['m01'] / M['m00']) cv2.circle(image, (cx, cy), 5, (255,0,0), -1) # بالاترین رقم = کمترین cy (چون مبدأ گوشه‌ی بالا-چپ است) highest_contour = min(contours, key=y_cord_contour) highest_y = y_cord_contour(highest_contour) # همه‌ی ارقامی که تقریباً هم‌ارتفاع با بالاترین رقم هستند (ردیف بالا) top_contours = [] for c in contours: y = y_cord_contour(c) if abs(y - highest_y) < 30: # اگر مرکز این عدد حداکثر 30 پیکسل با مرکز بالاترین عدد اختلاف داشت، جزو ردیف بالاست top_contours.append(c) # از بین ردیف بالا، راست‌ترین = بیشترین cx right_top_contour = max(top_contours, key=x_cord_contour) x, y, w, h = cv2.boundingRect(right_top_contour) cv2.rectangle(image, (x, y), (x + w, y + h), (0,0,255), 1) plt.figure(figsize=(15, 5)) plt.subplot(131); plt.imshow(gray, cmap="gray"); plt.title("Original"); plt.axis("off") plt.subplot(132); plt.imshow(binary_image, cmap="gray"); plt.title("Binary"); plt.axis("off") plt.subplot(133); plt.imshow(image[..., ::-1]); plt.title("Centers + Top Right Digit"); plt.axis("off") plt.tight_layout() plt.show()
