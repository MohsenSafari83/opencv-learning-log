1. رویدادهای ماوس (11)

اول ببینیم OpenCV چه رویدادهایی برای ماوس تعریف کرده:

import cv2 events = [i for i in dir(cv2) if 'EVENT' in i] print(events)

یک برنامه‌ی خیلی ساده که با کلیک، یک دایره رسم می‌کند:

import numpy as np import cv2 # mouse callback function def draw_circle(event, x, y, flags, param): if event == cv2.EVENT_LBUTTONDOWN: cv2.circle(img, (x,y), 20, (150,0,200), -1) # یک تصویر سیاه، یک پنجره، و اتصال تابع به پنجره img = np.zeros((512,512,3), np.uint8) cv2.namedWindow('image') cv2.setMouseCallback('image', draw_circle) while(1): cv2.imshow('image', img) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows()

الگوی کلی callback ماوس

تابع callback همیشه ۵ پارامتر می‌گیرد: event (نوع رویداد)، x, y (مختصات لحظه‌ی رویداد)، flags و param (کمتر استفاده می‌شوند). این تابع با cv2.setMouseCallback('نام‌پنجره', تابع) به یک پنجره‌ی مشخص وصل می‌شود.

برنامه‌ای شبیه قلم Paint — با نگه‌داشتن دکمه و حرکت ماوس نقاشی می‌کند:

drawing = False # true if mouse is pressed def brush(event, x, y, flags, param): global drawing if event == cv2.EVENT_LBUTTONDOWN: drawing = True elif event == cv2.EVENT_MOUSEMOVE: if drawing == True: cv2.circle(img, (x,y), 5, (0,0,255), -1) elif event == cv2.EVENT_LBUTTONUP: drawing = False img = np.zeros((512,512,3), np.uint8) cv2.namedWindow('image') cv2.setMouseCallback('image', brush) while True: cv2.imshow('image', img) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows() plt.imshow(img[...,::-1])

تعویض رنگ قلم با صفحه‌کلید (کلیدهای r, g, b):

drawing = False color = (0,255,0) def brush(event, x, y, flags, param): global drawing, color if event == cv2.EVENT_LBUTTONDOWN: drawing = True elif event == cv2.EVENT_MOUSEMOVE: if drawing == True: cv2.circle(img, (x,y), 5, color, -1) elif event == cv2.EVENT_LBUTTONUP: drawing = False img = np.zeros((512,512,3), np.uint8) cv2.namedWindow('image') cv2.setMouseCallback('image', brush) while True: cv2.imshow('image', img) k = cv2.waitKey(1) & 0xFF if k == ord('b'): color = (255,0,0) elif k == ord('g'): color = (0,255,0) elif k == ord('r'): color = (0,0,255) elif k == 27: break cv2.destroyAllWindows() plt.imshow(img[...,::-1])

مشکل نسخه‌های بالا: اگر ماوس سریع حرکت کند، بین نقاط فاصله می‌افتد (چون فقط دایره‌ی نقطه‌ای رسم می‌شود، نه خط بین دو موقعیت متوالی). راه‌حل: رسم خط بین موقعیت قبلی و فعلی ماوس در هر حرکت:

img = np.zeros((512,512,3)) drawing = False ix, iy = 0, 0 def draw(event, x, y, flags, params): global ix, iy, drawing if event == cv2.EVENT_LBUTTONDOWN: drawing = True ix, iy = x, y if event == cv2.EVENT_MOUSEMOVE: if drawing == True: # به‌جای دایره، یک خط از نقطه‌ی قبلی به نقطه‌ی فعلی می‌کشد cv2.line(img, pt1=(ix,iy), pt2=(x,y), color=(255,255,255), thickness=3) ix, iy = x, y if event == cv2.EVENT_LBUTTONUP: drawing = False cv2.namedWindow("Window") cv2.setMouseCallback("Window", draw) while True: cv2.imshow("Window", img) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows() plt.imshow(img[...,::-1])

رسم مستطیل توپر با درگ‌کردن ماوس:

drawing = False start_point = (0, 0) def draw_rectangle(event, x, y, flags, param): global start_point, drawing if event == cv2.EVENT_LBUTTONDOWN: drawing = True start_point = x, y elif event == cv2.EVENT_MOUSEMOVE: if drawing == True: cv2.rectangle(img, start_point, (x,y), (0,255,0), -1) elif event == cv2.EVENT_LBUTTONUP: drawing = False img = np.zeros((512,512,3), np.uint8) cv2.namedWindow('image') cv2.setMouseCallback('image', draw_rectangle) while True: cv2.imshow('image', img) k = cv2.waitKey(1) & 0xFF if k == 27: break cv2.destroyAllWindows() plt.imshow(img[...,::-1])

در نسخه‌ی بالا، در طول درگ‌کردن هیچ پیش‌نمایشی دیده نمی‌شود — مستطیل فقط لحظه‌ی رهاکردن دکمه ظاهر می‌شود. راه‌حل: در رویداد EVENT_MOUSEMOVE، روی یک کپی موقت از تصویر رسم کن و نمایشش بده (نه روی تصویر اصلی) — این‌طوری تصویر اصلی دست‌نخورده می‌ماند تا واقعاً کلیک تمام شود.

start_point = (0, 0) points = [] drawing = False def annotate_image(event, x, y, flags, param): global start_point, points, drawing if event == cv2.EVENT_LBUTTONDOWN: start_point = (x, y) drawing = True elif event == cv2.EVENT_MOUSEMOVE: if drawing: # پیش‌نمایش روی یک کپی موقت، نه روی تصویر اصلی temp_image = image.copy() cv2.rectangle(temp_image, start_point, (x, y), (120,255,0), 2) cv2.imshow("image", temp_image) elif event == cv2.EVENT_LBUTTONUP: points.append([start_point, (x, y)]) cv2.rectangle(image, start_point, (x, y), (0,255,0), 3) cv2.imshow("image", image) drawing = False image = cv2.imread('images/input.jpg') clone = image.copy() cv2.namedWindow("image") cv2.setMouseCallback("image", annotate_image) while True: if drawing == False: cv2.imshow("image", image) key = cv2.waitKey(5) & 0xFF if key == ord("r"): image = clone.copy() points = [] elif key == 27: break cv2.destroyAllWindows() plt.imshow(image[...,::-1]) print(points)

و در نهایت، افزودن قابلیت Undo (کلید z) — با پاک‌کردن آخرین نقطه‌ی ثبت‌شده از لیست و رسم دوباره‌ی همه‌ی مستطیل‌های باقی‌مانده روی یک کپی تازه:

elif key == ord("z"): # شبیه ctrl+z image = clone.copy() points.pop() for p in points: cv2.rectangle(image, p[0], p[1], (0,255,0), 3)

2. افزودن Trackbar (12)

Trackbar یعنی یک اسلایدر روی پنجره‌ی OpenCV که با تکان‌دادنش یک تابع callback صدا زده می‌شود. مثال کلاسیک: کنترل نسبت ترکیب دو تصویر (blend) با یک اسلایدر:

import cv2 alpha_slider_max = 100 title_window = 'Linear Blend' def on_trackbar(val): alpha = val / alpha_slider_max beta = (1.0 - alpha) dst = cv2.addWeighted(src1, alpha, src2, beta, 0.0) cv2.imshow(title_window, dst) src1 = cv2.imread('images/class.vision.jpg') src2 = cv2.imread('images/opencv.jpg') cv2.namedWindow(title_window) trackbar_name = 'Alpha x %d' % alpha_slider_max cv2.createTrackbar(trackbar_name, title_window, 0, alpha_slider_max, on_trackbar) on_trackbar(0) # مقدار اولیه رو نمایش بده cv2.waitKey() cv2.destroyAllWindows()

برای اینکه نتیجه‌ی نهایی (آخرین مقدار اسلایدر) بعد از بستن پنجره هم در خود نوت‌بوک قابل مشاهده باشد، کافیست متغیرها را global کنیم:

def on_trackbar(val): global dst, alpha # برای دسترسی matplotlib بعد از بسته‌شدن پنجره alpha = val / alpha_slider_max beta = (1.0 - alpha) dst = cv2.addWeighted(src1, alpha, src2, beta, 0.0) cv2.imshow(title_window, dst) # ... (بقیه مثل قبل) ... plt.imshow(dst[...,::-1]) print("alpha:{}".format(alpha))

مثال دیگر: تغییر رنگ یک صفحه با سه Trackbar جدا (R, G, B) به‌همراه یک سوییچ روشن/خاموش:

import numpy as np import cv2 def nothing(x): # وقتی callback واقعی لازم نیست، یک تابع خالی کافیه pass img = np.zeros((300,512,3), np.uint8) cv2.namedWindow('image') cv2.createTrackbar('R', 'image', 0, 255, nothing) cv2.createTrackbar('G', 'image', 0, 255, nothing) cv2.createTrackbar('B', 'image', 0, 255, nothing) switch = 'OFF:ON' cv2.createTrackbar(switch, 'image', 0, 1, nothing) while True: cv2.imshow('image', img) if cv2.waitKey(1) & 0xFF == 27: break # مقدار فعلی هر Trackbar را در هر فریم می‌خوانیم r = cv2.getTrackbarPos('R', 'image') g = cv2.getTrackbarPos('G', 'image') b = cv2.getTrackbarPos('B', 'image') s = cv2.getTrackbarPos(switch, 'image') if s == 0: img[:] = 0 else: img[:] = [b, g, r] cv2.destroyAllWindows()
i
دو الگوی مختلف Trackbar

الگوی اول (blend) از یک callback واقعی استفاده می‌کند که با هر تغییر اسلایدر بلافاصله اجرا می‌شود. الگوی دوم به‌جای این، یک nothing خالی به‌عنوان callback می‌دهد و مقدار Trackbar را داخل حلقه‌ی اصلی با getTrackbarPos می‌خواند — برای وقتی مفید است که چند Trackbar با هم ترکیب می‌شوند و callback جداگانه برای هرکدام دردسرساز است.

import cv2 import numpy as np cap = cv2.VideoCapture(0) while True: ret, frame = cap.read() height, width, _ = frame.shape zeros = np.zeros([height,width], dtype="uint8") B, G, R = cv2.split(frame) image_green = cv2.merge([zeros, G, zeros]) image_red = cv2.merge([zeros, zeros, R]) cv2.imshow('Webcam Green', image_green) cv2.imshow('Webcam Red', image_red) if cv2.waitKey(1) & 0xFF == 27: break cap.release() cv2.destroyAllWindows()

5. ذخیره‌ی ویدیو (15)

FourCC یک کد ۴ بایتی است که کدک فشرده‌سازی ویدیو را مشخص می‌کند (لیست کامل در fourcc.org / ویکی‌پدیا). چند نمونه‌ی رایج بسته به سیستم‌عامل:

سیستم‌عاملکدک‌های رایج
Fedora / LinuxDIVX, XVID, MJPG, X264, WMV1, WMV2
WindowsDIVX
macOSMJPG (.mp4), DIVX (.avi), X264 (.mkv)

نکته: XVID معمولاً ترجیح داده می‌شود؛ MJPG حجم فایل خروجی را زیاد می‌کند؛ X264 حجم بسیار کمی تولید می‌کند.

برای ساخت یک VideoWriter باید مشخص کنید: نام فایل خروجی، کد FourCC، تعداد فریم بر ثانیه (fps)، ابعاد فریم، و اینکه ویدیو رنگی است یا نه (isColor):

fourcc = cv2.VideoWriter_fourcc(*'MJPG') # یا معادلش: cv2.VideoWriter_fourcc('M','J','P','G')

ذخیره‌ی ویدیوی وب‌کم (آینه‌شده) در یک فایل:

import numpy as np import cv2 cap = cv2.VideoCapture(0) # تعریف کدک و ساخت شیء VideoWriter fourcc = cv2.VideoWriter_fourcc(*'XVID') out = cv2.VideoWriter('output.avi', fourcc, 20.0, (640, 480)) while cap.isOpened(): ret, frame = cap.read() if not ret: print("Can't receive frame (stream end?). Exiting ...") break frame = cv2.flip(frame, 1) out.write(frame) # نوشتن فریم آینه‌شده در فایل خروجی cv2.imshow('frame', frame) if cv2.waitKey(1) == ord('q'): break # در پایان، هم دوربین و هم فایل خروجی باید release بشن cap.release() out.release() cv2.destroyAllWindows()

خواندن یک ویدیوی موجود، تبدیل به گری‌اسکیل، و ذخیره‌ی دوباره در یک فایل جدید:

cap = cv2.VideoCapture('./videos/race_car.mp4') # عرض و ارتفاع باید عدد صحیح باشن w = int(cap.get(3)) h = int(cap.get(4)) # آرگومان آخر (False) یعنی ویدیوی خروجی گری‌اسکیله (تک‌کاناله) out = cv2.VideoWriter('race_car_gray.avi', cv2.VideoWriter_fourcc('M','J','P','G'), cap.get(cv2.CAP_PROP_FPS), (w, h), False) while True: ret, frame = cap.read() if ret == True: gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY) out.write(gray) else: print("Done") break cap.release() out.release() cv2.destroyAllWindows()
!
isColor باید با نوع فریم هم‌خوانی داشته باشد

چون فریم‌های نوشته‌شده گری‌اسکیل (تک‌کاناله) هستند، پارامتر isColor باید False باشد؛ در غیر این صورت VideoWriter انتظار فریم سه‌کاناله دارد و خروجی خراب یا خطا می‌دهد.

6. دوربین مداربسته و RTSP (16)

دوربین‌های مداربسته (CCTV/IP Camera) معمولاً یک آدرس استریم RTSP می‌دهند. برای OpenCV، این آدرس دقیقاً مثل مسیر یک فایل ویدیویی به VideoCapture داده می‌شود:

import cv2 # یک لینک تست رایگان RTSP (Our Free Test RTSP Link) # می‌تونید دوربین CCTV/IP خودتون رو هم طوری تنظیم کنید که یک استریم RTSP بده cap = cv2.VideoCapture("rtsp://wowzaec2demo.streamlock.net/vod/mp4:BigBuckBunny_115k.mp4") while True: ret, frame = cap.read() cv2.imshow('CCTV', frame) if cv2.waitKey(1) == 27: break cap.release() cv2.destroyAllWindows()

اگر بخواهید هم‌زمان چند دوربین مداربسته را در پایتون کنترل کنید، کتابخانه‌ی imagezmq جایگزین خوبی است — هر دستگاه (مثلاً چند Raspberry Pi) فریم‌هایش را از طریق ZMQ برای یک هاب مرکزی می‌فرستد:

import cv2 import imagezmq image_hub = imagezmq.ImageHub() while True: # show streamed images until Ctrl-C rpi_name, image = image_hub.recv_image() cv2.imshow(rpi_name, image) # یک پنجره برای هر دستگاه cv2.waitKey(1) image_hub.send_reply(b'OK')

import numpy as np from PIL import ImageGrab import cv2 while True: frame = np.array(ImageGrab.grab()) # نمایش صفحه (چون ImageGrab بر خلاف OpenCV به‌صورت RGB می‌ده) cv2.imshow('Screen', cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows()

screen_img = ImageGrab.grab() screen_img.save('screen.jpg')

while True: frame = np.array(ImageGrab.grab(bbox=(500,500,1000,1000))) cv2.imshow('Screen area', cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows()

import numpy as np import cv2 from mss import mss bounding_box = {'top': 100, 'left': 50, 'width': 400, 'height': 300} sct = mss() while True: sct_img = sct.grab(bounding_box) cv2.imshow('screen', np.array(sct_img)) if cv2.waitKey(1) & 0xFF == 27: break cv2.destroyAllWindows()

import pafy import cv2 url = 'https://www.youtube.com/watch?v=-VtXTkvDRW8' video = pafy.new(url) best = video.getbest(preftype="mp4") capture = cv2.VideoCapture() capture.open(best.url) while (True): ret, frame = capture.read() if ret == True: cv2.imshow('src', frame) if cv2.waitKey(1) == 13: # 13 یعنی کلید Enter break capture.release() cv2.destroyAllWindows()

جمع‌بندی ماژول 2

  • cv2.setMouseCallback رویدادهای ماوس (کلیک، حرکت، رهاکردن) را به یک تابع callback با امضای ثابت (event, x, y, flags, param) وصل می‌کند.
  • برای preview زنده‌ی یک ترسیم در حال کشیدن‌شدن (قبل از پایان کلیک)، همیشه روی یک کپی موقت از تصویر رسم کنید، نه روی تصویر اصلی.
  • Trackbar یا یک callback واقعی می‌گیرد (اجرا با هر تغییر مقدار)، یا با یک callback خالی (nothing) ساخته می‌شود و مقدارش با getTrackbarPos در حلقه خوانده می‌شود.
  • cv2.VideoCapture منبع ورودی می‌تواند وب‌کم (0)، مسیر فایل، آدرس RTSP، یا حتی آدرس مستقیم استریم یوتیوب (از طریق pafy) باشد — رابط خواندنش همیشه یکسان است: ret, frame = cap.read().
  • برای فایل (برخلاف وب‌کم)، چک‌کردن if not ret: break ضروری است چون ویدیو پایان دارد.
  • cv2.VideoWriter برای ذخیره‌سازی نیاز به کدک FourCC، fps، ابعاد فریم، و پرچم رنگی/گری‌اسکیل (isColor) دارد که باید با نوع واقعی فریم‌های نوشته‌شده هم‌خوانی داشته باشد.
  • ضبط صفحه‌نمایش (ImageGrab یا سریع‌تر: mss) هم دقیقاً با همان الگوی حلقه‌ی imshow قابل زنده‌نمایی یا ذخیره‌سازی است.
