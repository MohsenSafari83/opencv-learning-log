تمرین 1 — رسم مثلث با کلیک ماوس

یک برنامه‌ی ساده بنویسید که با کلیک کاربر روی صفحه، در نقطه‌ی کلیک‌شده یک مثلث رسم کند. رنگ این مثلث‌ها باید با دکمه‌های روی کیبورد قابل‌تغییر باشد.

q5.ipynbرسم مثلث با کلیک ماوسPY
import cv2 import numpy as np import matplotlib.pyplot as plt color = (0, 255, 0) # رنگ پیش‌فرض: سبز def draw_triangle(event, x, y, flags, param): global color if event == cv2.EVENT_LBUTTONDOWN: cx, cy = x, y # نقطه‌ی کلیک = مرکز مثلث r = 40 # اندازه‌ی مثلث # سه رأس مثلث حول مرکز کلیک pts = np.array([ [cx, cy - r], [cx - r, cy + r], [cx + r, cy + r] ], np.int32) cv2.fillPoly(image, [pts], color) image = np.ones((600, 800, 3), dtype=np.uint8) * 255 cv2.namedWindow("image") cv2.setMouseCallback("image", draw_triangle) while True: cv2.imshow("image", image) key = cv2.waitKey(1) & 0xFF if key == ord("r"): color = (0, 0, 255) elif key == ord("g"): color = (0, 255, 0) elif key == ord("b"): color = (255, 0, 0) elif key == 27: break cv2.destroyAllWindows() plt.imshow(image[...,::-1])

تمرین 2 — یک برنامه‌ی نقاشی ساده (Mini Paint)

یک برنامه‌ی ساده شبیه Paint بنویسید که بتوان رنگ قلم و اندازه‌ی قلم را تغییر داد. برای تغییر اندازه و رنگ قلم از Trackbar استفاده کنید — ۴ تا Trackbar داشته باشید که به‌ترتیب سه رنگ R، G، B و اندازه‌ی قلم را مشخص کنند.

q6.ipynbنقاشی با TrackbarPY
import matplotlib.pyplot as plt import numpy as np import cv2 img = np.zeros((512, 512, 3), dtype=np.uint8) drawing = False ix = 0 iy = 0 def draw(event, x, y, flags, params): global ix, iy, drawing if event == cv2.EVENT_LBUTTONDOWN: drawing = True ix = x iy = y elif event == cv2.EVENT_MOUSEMOVE: if drawing == True: # رنگ و اندازه‌ی قلم هر بار از Trackbarها خونده می‌شن r = cv2.getTrackbarPos("R", "Window") g = cv2.getTrackbarPos("G", "Window") b = cv2.getTrackbarPos("B", "Window") size = cv2.getTrackbarPos("Size", "Window") if size == 0: size = 1 # جلوگیری از خط با ضخامت صفر color = (b, g, r) cv2.line( img, pt1=(ix, iy), pt2=(x, y), color=color, thickness=size ) ix = x iy = y elif event == cv2.EVENT_LBUTTONUP: drawing = False cv2.namedWindow("Window") cv2.createTrackbar("R", "Window", 255, 255, lambda x: None) cv2.createTrackbar("G", "Window", 255, 255, lambda x: None) cv2.createTrackbar("B", "Window", 255, 255, lambda x: None) cv2.createTrackbar("Size", "Window", 3, 50, lambda x: None) cv2.setMouseCallback("Window", draw) while True: cv2.imshow("Window", img) key = cv2.waitKey(1) & 0xFF if key == ord("c"): # پاک‌کردن کل بوم img[:] = 0 elif key == 27: break cv2.destroyAllWindows() plt.imshow(img[..., ::-1]) plt.axis("off") plt.show()

تمرین 3 — تصحیح گاما زنده روی تصویر وب‌کم

یک برنامه بنویسید که تصویر اصلی وب‌کم و همچنین تصویر وب‌کم بعد از اعمال تصحیح روشنایی و کنتراست (به روش تصحیح گاما) را هم‌زمان نمایش دهد. این برنامه با فشردن q روی کیبورد باید بسته شود.

q7.ipynbتصحیح گاما زنده روی وب‌کمPY
import cv2 import numpy as np cap = cv2.VideoCapture(0) if not cap.isOpened(): print("can't open webcam") exit() gamma = 0.5 # فرمول درست: O = 255 * (I/255)^gamma — گاما مستقیم تو توان می‌ره، نه معکوسش # (اگه به‌جای gamma از 1.0/gamma استفاده کنی، برای gamma=0.5 عملاً # معادل گامای 2 اجرا می‌شه و تصویر به‌جای روشن‌تر، تیره‌تر می‌شه) table = np.array([ ((i / 255.0) ** gamma) * 255 for i in np.arange(256) ]).astype("uint8") while True: ret, frame = cap.read() if not ret: print("Can't receive frame (stream end?). Exiting ...") break original = frame corrected = cv2.LUT(original, table) cv2.imshow("Original Webcam", original) cv2.imshow("Gamma Corrected", corrected) if cv2.waitKey(1) & 0xFF == ord("q"): break cap.release() cv2.destroyAllWindows()
