تمرین 1 — برش تصویر (چهره)

تصویر یک فرد دلخواه لود شود، با نمایش در matplotlib و مشاهده‌ی محور مختصات، ناحیه‌ی چهره برش زده و در یک فایل جدید ذخیره شود.

import cv2 import numpy as np import matplotlib.pyplot as plt import os image_path = os.path.join('images', 'tusk.jpg') # مسیر نسبی به‌جای مسیر ثابت image = cv2.imread(image_path) assert image is not None, f"تصویر پیدا نشد: {image_path}" plt.imshow(image[...,::-1]) # با مشاهده‌ی محورها، مختصات دقیق ناحیه‌ی چهره پیدا شده y1, y2, x1, x2 = 120, 320, 235, 390 cropped = image[y1:y2, x1:x2] # بررسی اینکه برش خالی یا خارج از ابعاد تصویر نباشد assert cropped.size > 0, "ناحیه‌ی برش خالی یا خارج از ابعاد تصویر است" plt.imshow(cropped[...,::-1]) output_path = os.path.join('images', 'face.jpg') cv2.imwrite(output_path, cropped) cv2.imshow("Original", image) cv2.imshow("Cropped", cropped) cv2.waitKey(0) cv2.destroyAllWindows()

import cv2 import numpy as np import matplotlib.pyplot as plt image = cv2.imread("tusk.jpg") image_bgra = cv2.cvtColor(image, cv2.COLOR_BGR2BGRA) # افزودن کانال آلفا (کانال چهارم) # مقدار آلفای کمتر از 255 باعث ایجاد حالت شیشه‌ای در پس‌زمینه می‌شود alpha = np.full(image.shape[:2], 80, dtype=np.uint8) alpha[120:320, 235:390] = 255 # ناحیه‌ی چهره کاملاً مات/دیدنی image_bgra[:, :, 3] = alpha cv2.imwrite("face_transparent.png", image_bgra) plt.figure(figsize=(6,6)) plt.imshow(cv2.cvtColor(image_bgra, cv2.COLOR_BGRA2RGBA)) plt.axis("off") plt.show()

تمرین 3 — رسم یک خانه‌ی ساده

با رسم مستطیل و مثلث، یک خانه‌ی ساده (مانند نقاشی کودکان) رسم شود.

q3.ipynbرسم خانه با اشکال پایهPY
import cv2 import numpy as np import matplotlib.pyplot as plt img = np.ones((500, 500, 3), dtype=np.uint8) * 255 # بوم سفید # یک تابع کمکی برای اینکه همه‌ی نماها یک الگوی یکدست (پر + کادر) داشته باشند def filled_shape_with_border(img, pts_or_rect, fill_color, border_color=(0,0,0), thickness=2, is_poly=False): if is_poly: cv2.fillPoly(img, [pts_or_rect], fill_color) cv2.polylines(img, [pts_or_rect], True, border_color, thickness) else: (x1, y1), (x2, y2) = pts_or_rect cv2.rectangle(img, (x1, y1), (x2, y2), fill_color, -1) cv2.rectangle(img, (x1, y1), (x2, y2), border_color, thickness) # بدنه‌ی خانه filled_shape_with_border(img, [(150, 250), (350, 450)], (100, 180, 255)) # سقف مثلثی — همان الگو: fillPoly برای سطح + polylines برای مرز triangle = np.array([[120, 250], [250, 100], [380, 250]], np.int32).reshape((-1, 1, 2)) filled_shape_with_border(img, triangle, (60, 60, 200), border_color=(0, 0, 255), thickness=5, is_poly=True) # در خانه filled_shape_with_border(img, [(220, 330), (280, 450)], (50, 50, 50)) # پنجره‌ها — همان الگو دوباره استفاده می‌شود filled_shape_with_border(img, [(170, 300), (210, 340)], (255, 255, 0)) filled_shape_with_border(img, [(290, 300), (330, 340)], (255, 255, 0)) plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)) plt.axis("off") plt.show()

تمرین 4 — روشن‌کردن یک تصویر تاریک (خطی در برابر گاما)

تصویر Weeki Wachee Spring که خیلی تاریک است، یک‌بار با روش خطی و یک‌بار با تصحیح گاما روشن‌تر شود (راهنما: صفحه‌ی گاما در ویکی‌پدیا).

q4.ipynbمقایسه‌ی روش خطی و گاماPY
import cv2 import numpy as np import matplotlib.pyplot as plt import os image_path = os.path.join('images', 'test.jpg') # مسیر نسبی به‌جای مسیر ثابت img = cv2.imread(image_path) assert img is not None, f"تصویر پیدا نشد: {image_path}" def linear_transform(image, alpha, beta): return cv2.convertScaleAbs(image, alpha=alpha, beta=beta) # y = 255 * (x/255)^gamma def gamma_correction(image, gamma): table = np.array([ ((i / 255.0) ** gamma) * 255 for i in np.arange(256) ]).astype("uint8") return cv2.LUT(image, table)
gammas = [2, 1, 1/2, 1/3, 1/4] linear_params = [ (1.0, 60), (1.3, 45), (1.6, 30), (2.0, 15), (2.5, 0) ] output_dir = 'outputs' os.makedirs(output_dir, exist_ok=True) plt.figure(figsize=(12, 15)) for i in range(len(gammas)): alpha, beta = linear_params[i] linear_img = linear_transform(img, alpha, beta) gamma_img = gamma_correction(img, gammas[i]) # ذخیره‌ی خروجی نهایی روی دیسک cv2.imwrite(os.path.join(output_dir, f"linear_{i}.jpg"), linear_img) cv2.imwrite(os.path.join(output_dir, f"gamma_{i}.jpg"), gamma_img) plt.subplot(5, 2, i*2+1) plt.imshow(cv2.cvtColor(linear_img, cv2.COLOR_BGR2RGB)) plt.title(f"Linear: α={alpha}, β={beta}") plt.axis("off") plt.subplot(5, 2, i*2+2) plt.imshow(cv2.cvtColor(gamma_img, cv2.COLOR_BGR2RGB)) plt.title(f"Gamma = {gammas[i]}") plt.axis("off") plt.tight_layout() plt.show()
